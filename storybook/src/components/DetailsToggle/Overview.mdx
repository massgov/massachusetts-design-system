import { renderDetailsToggle } from '../../utils/component-renderers.js';
import * as DetailsToggleStories from './details-toggle.stories.js';

export const DetailsToggleExample = ({ open = false, ...args }) => {
  const markup = renderDetailsToggle({
    ...detailsToggleDefaults,
    ...args
  });

  return (
    <div
      className="sb-unstyled mds-docs-example-frame"
      dangerouslySetInnerHTML={{
        __html: open ? markup.replace('<details ', '<details open ') : markup
      }}
    />
  );
};

<Meta of={DetailsToggleStories} />

# Details toggle overview

Details toggle lets people show or hide additional supporting information without adding unnecessary content to the page. Use it for optional explanations that help people understand or complete a task.

Keep essential instructions visible without requiring people to expand the toggle.
